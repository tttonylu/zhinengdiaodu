## 证据采集辅助（控制台粘贴）

```js
(() => {
  let last = '';
  const pick = (d) => {
    const cd = (d && d.chatDiag) || {}, s = cd.softRisk || {};
    const running = d.followRunning ? 'follow' : d.chatRunning ? 'chat' : d.nurtureRunning ? 'nurture' : d.repostRunning ? 'repost' : '-';
    return [
      'stage=' + (d.stage ?? '-'),
      'engine=' + (d.engine ?? '-'),
      'run=' + running,
      'fPhase=' + ((d.moduleLive && d.moduleLive.follow && d.moduleLive.follow.phase) || '-'),
      'fRound=' + (cd.followRoundActive ? 'T' : 'F'),
      'soft=' + (s.active ? (s.mode || '?') + '/' + (s.remainSec ?? '?') + 's' : '-'),
      'gate=' + (cd.opsGateReason || '-'),
      'debt=' + (cd.actionableAdDebt ?? 0) + '/' + (cd.actionableKickDebt ?? 0)
    ].join(' | ');
  };
  console.log('%c[sg-accept] 证据采集开始（10s/次）', 'color:#3fb950');
  const timer = setInterval(async () => {
    try {
      const d = await window.__sgDiag();
      if (!d || d.error) return;
      const line = pick(d);
      if (line !== last) { console.log('%c' + new Date().toLocaleTimeString() + ' ▸ ' + line, 'color:#f0883e'); last = line; }
    } catch (e) {}
  }, 10000);
  window.__sgAcceptStop = () => { clearInterval(timer); console.log('%c[sg-accept] 采集停止', 'color:#f85149'); };
})();
```
